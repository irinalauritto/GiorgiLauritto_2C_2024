168static void calcularParametrosECG(void *pvParameter){
169 while (true)
170 {
171 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
172 PROCESANDO = true;
173 uint16_t umbralVoltaje = 350;
174 uint16_t posicionAnteriorQRS = 0;
175 float periodoCardiaco;
176
177 // Cálculo de frecuencia cardíaca
178 uint8_t contadorQRS = 0;
179
180 uint16_t j = 0;
181
182 while (j < BUFFER_SIZE)
183 {
184 if (ecg[j] > umbralVoltaje)
185 {
186 contadorQRS++;
187 printf("contador: %d\n", contadorQRS);
188 float frecuenciaAUX = frecuenciaCardiaca;
189 uint16_t deltaPosicion = j - posicionAnteriorQRS;
190 posicionAnteriorQRS = j;
191 periodoCardiaco = deltaPosicion * 2.5 / BUFFER_SIZE; // delta en tiempo
192 frecuenciaCardiaca = 1 / (periodoCardiaco/60);
193
194 if(contadorQRS==1){
195 printf("frec:%.2f\n",frecuenciaAUX);
196 frecuenciaCardiaca=frecuenciaAUX;
197 }
198 else{
199 printf("frec:%.2f\n", frecuenciaCardiaca);
200 }
201
202
203 //Compara si tiene taquicardia o no:
204 if (frecuenciaCardiaca > limiteTaquicardia)
205 {
206 TAQUICARDIA = true;
207
208 }
209 else
210 {
211 TAQUICARDIA = false;
212 }
213 // Comparación con bradicardia
214 if (frecuenciaCardiaca < limiteBradicardia)
215 {
216 BRADICARDIA = true;
217 }
218 else
219 {
220 BRADICARDIA = false;
221 }
222 if(limiteBradicardia<frecuenciaCardiaca && frecuenciaCardiaca<limiteTaquicardia)
223 {
224 FRECUENCIA_NORMAL = true;
225 }
226 else
227 {
228 FRECUENCIA_NORMAL = false;
229 }
230
231 j = j + 70;
232 }
233 j++;
234 }
235
236 PROCESANDO = false;

273static void mostrar(void *pvParameter){
274 char msg[128];
275 char msg_chunk[24];
276 char frecuencia[128];
277 static uint8_t indice = 0;
278 while(true){
279 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
280 if(filter){
281 HiPassFilter(&ecg[indice], ecg_filt, CHUNK);
282 LowPassFilter(ecg_filt, ecg_filt, CHUNK);
283 } else{
284 memcpy(ecg_filt, &ecg[indice], CHUNK*sizeof(float));
285 }
286 strcpy(msg, "");
287 for(uint8_t i=0; i<CHUNK; i++){
288 sprintf(msg_chunk, "*G%.2f*", ecg_filt[i]);
289 strcat(msg, msg_chunk);
290 }
291 indice += CHUNK;
292
293 BleSendString(msg);
294
295 }

340void app_main(void){
341
342 ble_config_t ble_configuration = {
343 "ESP_EDU_IRI_JOSE",
344 read_data

347 timer_config_t timer_senial = {

351 .param_p = NULL

353
354

358 HiPassInit(SAMPLE_FREQ, 1, ORDER_2);
359
360 BleInit(&ble_configuration);
361
362 timer_config_t timerECG = {
363 .timer = TIMER_A,
364 .period = RETARDO_ECG,
365 .func_p = funcTimerECG,
366 .param_p = NULL
367 };
368 TimerInit(&timerECG);
369
370 // Inicialización del Convertidor AD
371 analog_input_config_t convertidorAD = {
372 .input = CH1,
373 .mode = ADC_SINGLE,
374 };
375 AnalogInputInit(&convertidorAD);
376 AnalogOutputInit();
377
378 xTaskCreate(&adquirirProcesarECG, "adquirirProcesarECG", 2048, NULL, 5, &adquirirProcesarECGTaskHandle);
379 xTaskCreate(&calcularParametrosECG, "calcularParametrosECG", 2048, NULL, 5, &calcularParametrosECGTaskHandle);
380 xTaskCreate(&mostrar, "mostrar", 2048, NULL, 5, &mostrarTaskHandle);
381 xTaskCreate(&mandarDiagnostico, "mandarDiagnostico", 2048, NULL, 5, &mandarDiagnosticoTaskHandle);
382
383 //Inicialización del puerto serie
384 serial_config_t myUart = {
385 .port = UART_PC,
386 .baud_rate = 115200,
387 .func_p = NULL,
388 .param_p = NULL,
389 };
390 UartInit(&myUart);
391
392 // Inicialización del conteo de timers
393 TimerStart(timerECG.timer);
394 TimerStart(timer_senial.timer);
395
396
397 while(1){
398 vTaskDelay(CONFIG_BLINK_PERIOD / portTICK_PERIOD_MS);
399 switch(BleStatus()){
400 case BLE_OFF:
401 LedOff(LED_BT);
402 break;
403 case BLE_DISCONNECTED:
404 LedToggle(LED_BT);
405 break;
406 case BLE_CONNECTED:
407 LedOn(LED_BT);
408 break;
409 }
410 }
